\question Using a dictionary comprehension, complete the function below. It takes a single-argument function \lstinline{f} and a dictionary \lstinline{d}, and returns a new dictionary containing only the entries whose keys are even numbers, with \lstinline{f} applied to their values.

    
    
\begin{lstlisting}
def apply_to_even(f, d):
    '''
    >>> apply_to_even(lambda x: x**2, {1:3, 2:5, 4:2, 3:2})
    {2: 25, 4: 4}
    '''
    return {__ : _____ for ____ in ____ if ___________}
\end{lstlisting}

\begin{solution}
\begin{lstlisting}
def apply_to_even(f, d):
    return {i: f(d[i]) for i in d if i % 2 == 0}
\end{lstlisting}
\end{solution}
