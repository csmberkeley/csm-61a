\begin{blocksection}
\question Write a query that returns all columns for the three chairs with the most single people. Break ties by putting the smaller chair number first.

\begin{solution}[1.5in]
\begin{lstlisting}
SELECT * FROM lift
ORDER BY single DESC, chair ASC
LIMIT 3;
\end{lstlisting}
\end{solution}
\end{blocksection}
