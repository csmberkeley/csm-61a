\begin{blocksection}
\question Assume \lstinline{lift} has integer columns \lstinline{chair}, \lstinline{single}, and \lstinline{couple}. Write a query that returns the chair numbers for chairs with more couples than single people.

\begin{solution}[1.5in]
\begin{lstlisting}
SELECT chair FROM lift WHERE couple > single;
\end{lstlisting}
\end{solution}
\end{blocksection}
