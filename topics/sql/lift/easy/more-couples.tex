\begin{blocksection}
\question Assume \lstinline{lift} has the above schema. Write a query that returns the chair number of every chair where the number of couples is greater than the number of single people. (Compare the \lstinline{couple} and \lstinline{single} columns directly; don't count each couple as two people here.)

\begin{solution}[1.5in]
\begin{lstlisting}
SELECT chair FROM lift WHERE couple > single;
\end{lstlisting}
\end{solution}
\end{blocksection}
