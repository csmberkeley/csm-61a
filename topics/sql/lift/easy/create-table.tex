\question Write a \lstinline{CREATE TABLE} statement for a table named \lstinline{lift} with three integer columns: \lstinline{chair}, \lstinline{single}, and \lstinline{couple}.

\begin{solution}[2in]
\begin{lstlisting}
CREATE TABLE lift (
    chair INTEGER,
    single INTEGER,
    couple INTEGER
);
\end{lstlisting}
\end{solution}
