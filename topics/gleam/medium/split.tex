\begin{solution}
\begin{lstlisting}
pub fn split(l: LinkedList(a), at: a) -> #(LinkedList(a), LinkedList(a)) {
  case l {
    Empty -> #(Empty, Empty)
    Link(first, _) if first == at -> #(Empty, l)
    Link(first, rest) -> {
      let #(before, after) = split(rest, at)
      #(Link(first, before), after)
    }
  }
}
\end{lstlisting}
\end{solution}
