pub fn prune_even(tree: Tree(Int)) -> Tree(Int) {
    let Tree(label, branches) = tree

    let remove_even =
        list.filter(branches, fn(branch) { branch.label % 2 != 0 })
    let prune_children = list.map(remove_even, prune_even)

    Tree(label, prune_children)
}
