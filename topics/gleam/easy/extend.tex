\begin{blocksection}
\question Implement \texttt{extend}, which takes in two LinkedLists \texttt{a} and \texttt{b} and returns a LinkedList with the elements of \texttt{b} after the elements of \texttt{a}.
\begin{lstlisting}
pub fn extend(a: LinkedList(t), b: LinkedList(t)) -> LinkedList(t) {
  case a {
    Empty -> ___________
    Link(first, rest) -> ____________________________________
  }
}
\end{lstlisting}

\begin{solution}
\begin{lstlisting}
pub fn extend(a: LinkedList(t), b: LinkedList(t)) -> LinkedList(t) {
  case a {
    Empty -> b
    Link(first, rest) -> Link(first, extend(rest, b))
  }
}
\end{lstlisting}
\end{solution}
