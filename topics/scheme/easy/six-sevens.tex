\begin{blocksection}
\question Define \lstinline$six-sevens$. Return the number of times that \lstinline$7$
follows \lstinline$6$ in the list.

\begin{lstlisting}
> (six-sevens '(4 6 7 6 0 7))
1
> (sixty-ones '(7 6 7 4 6 7 6 0 7))
2
> (sixty-ones '(6 7 6 7 4 6 7 6 0 7))
3

(define (six-sevens lst)
    (cond (____________________________________________)
          (_______________________________________________________________________)
          (else ______________________________)))
\end{lstlisting}

\begin{solution}[1.5in]
\begin{lstlisting}
(define (six-sevens lst)
    (cond ((or (null? lst) (null? (cdr lst))) 0)
          ((and (= 6 (car lst)) (= 7 (car (cdr lst)))) (+ 1 (six-sevens (cdr(cdr lst)))))
          (else (six-sevens (cdr lst)))))
\end{lstlisting}
\end{solution}
\end{blocksection}
