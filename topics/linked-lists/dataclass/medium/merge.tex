\question Write \lstinline{merge}, which takes two linked lists of integers, each sorted in increasing order, and returns a new sorted linked list containing all of the elements of both.

\begin{lstlisting}

    """
    >>> a = Link(1, Link(5))
    >>> b = Link(2, Link(3, Link(8)))
    >>> print(merge(a, b))
    (1 2 3 5 8)
    >>> print(merge(a, ()))
    (1 5)
    >>> print(merge(Link(1, Link(3)), Link(1)))
    (1 1 3)
    >>> merge((), ())
    ()
    """

        return ______________________________

    elif ____________________________________:

\end{lstlisting}

\begin{solution}[1in]
\begin{lstlisting}
def merge(a: LinkedList[int], b: LinkedList[int]) -> LinkedList[int]:
    if not isinstance(a, Link):
        return b
    elif not isinstance(b, Link):
        return a
    elif a.first <= b.first:
        return Link(a.first, merge(a.rest, b))
    else:
        return Link(b.first, merge(a, b.rest))
\end{lstlisting}
\end{solution}

\begin{questionmeta}
\textbf{Teaching Tips}\par\nopagebreak[4]
Suggested Time: 10 min; Difficulty: Medium
\nopagebreak[4]
\begin{itemize}
    \item \textbf{Base cases:} If either list is empty, the answer is the other list (which may also be empty).
    \item \textbf{Smallest first:} The smallest remaining element is \lstinline{a.first} or \lstinline{b.first}. Put it on the front, then merge the remaining elements.
    \item \textbf{Which arguments shrink:} Only the list whose first element was used gets shorter; the other is passed unchanged.
    \item \textbf{Duplicates:} In \lstinline{merge(Link(1, Link(3)), Link(1))}, both lists start with 1. Both copies must appear in the result, so a solution must not drop or skip equal elements.
\end{itemize}
\end{questionmeta}
