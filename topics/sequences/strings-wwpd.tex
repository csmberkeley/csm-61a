\begin{blocksection}
    \question What would Python display?
    
    \begin{lstlisting}
    >>> a = "hello world!"
    >>> a[4]
    \end{lstlisting}
    \begin{solution}[.25in]
    \begin{lstlisting}
    'o'
    \end{lstlisting}
    \end{solution}

    \begin{lstlisting}
    >>> a = "hello world!"
    >>> a[6:11]
    \end{lstlisting}
    \begin{solution}[.25in]
    \begin{lstlisting}
    'world'
    \end{lstlisting}
    \end{solution}

    \begin{lstlisting}
    >>> len(a)
    \end{lstlisting}
    \begin{solution}[.25in]
    \begin{lstlisting}
    12
    \end{lstlisting}
    \end{solution}

    \begin{lstlisting}
    >>> 'lo' in a
    \end{lstlisting}
    \begin{solution}[.25in]
    \begin{lstlisting}
    True
    \end{lstlisting}
    \end{solution}

    
    \begin{questionmeta}
        \textbf{Teaching Tips}
        \begin{enumerate}
                \item The goal of this problem is to introduce dictionaries. 
                \item Encourage students to draw box and pointer diagrams if they seem stuck, but it is not necessary to draw a box and pointer diagram for all questions. 
        \end{enumerate}
    \end{questionmeta}
\end{blocksection}
