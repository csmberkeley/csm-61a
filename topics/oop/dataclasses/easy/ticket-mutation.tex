\begin{blocksection}
\question \textbf{Museum tickets.} Predict each result, executing the prompts
in order. Draw the instances and the references to them as you trace.

\begin{lstlisting}
from dataclasses import dataclass

@dataclass
class Ticket:
    exhibit: str
    remaining: int = 2

    def admit(self):
        if self.remaining > 0:
            self.remaining -= 1
            return True
        return False

original = Ticket("Space")
alias = original
separate = Ticket("Space", 1)
\end{lstlisting}
\end{blocksection}
