\begin{blocksection}
\question Rewrite each function in the requested equivalent form. Preserve the
function's parameters and behavior; only change whether it is written with
\lstinline{def} or \lstinline{lambda}. For your reference, a \lstinline{lambda} assignment assigns a lambda function to a variable.

\begin{parts}
\part Rewrite the following \lstinline{def} as a lambda assignment.
\begin{lstlisting}
def square(x):
    return x * x
\end{lstlisting}

\begin{solution}[0.25in]
\lstinline$square = lambda x: x * x$
\end{solution}

\part Rewrite the following lambda assignment using \lstinline{def}.
\begin{lstlisting}
subtract = lambda x, y: x - y
\end{lstlisting}

\begin{solution}[0.45in]
\begin{lstlisting}
def subtract(x, y):
    return x - y
\end{lstlisting}
\end{solution}

\part Rewrite this zero-argument function as a lambda assignment.
\begin{lstlisting}
def greeting():
    return 'Hello, world!'
\end{lstlisting}

\begin{solution}[0.25in]
\lstinline$greeting = lambda: 'Hello, world!'$
\end{solution}
\end{parts}
\end{blocksection}
