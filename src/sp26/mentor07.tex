\section{Trees}
\begin{questions}
    \subimport{../../topics/trees/adt/easy/}{height.tex}
    \begin{questionmeta}
        Suggested Time: 7 min; Difficulty: Easy
        \begin{itemize}
            \item  Good fundemental question that goes over how tree recursion works. Think about why the base case would be isleaf, and how we can ensure the height takes into account the tallest subtree in a tree. 
        \end{itemize}
      \end{questionmeta}    
      
      \subimport{../../topics/trees/adt/easy/}{tree-sum.tex}
      \begin{questionmeta}
          Suggested Time: 7 min; Easy
          \begin{itemize}
              \item  Also fundemental question that tests how to do tree recursion without returning the recursive case. How can we take advantage of the qualities of a tree? 
          \end{itemize}
        \end{questionmeta}

        \newpage

        \subimport{../../topics/trees/adt/medium/}{weighted-tree.tex}

            Suggested Time: 12 min; Medium
